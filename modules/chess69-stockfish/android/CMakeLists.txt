cmake_minimum_required(VERSION 3.22.1)
project(chess69_stockfish LANGUAGES CXX)
set(STOCKFISH ${CMAKE_CURRENT_SOURCE_DIR}/../../../android/vendor/stockfish/src)
file(GLOB STOCKFISH_SOURCES ${STOCKFISH}/*.cpp ${STOCKFISH}/nnue/*.cpp ${STOCKFISH}/nnue/features/*.cpp ${STOCKFISH}/syzygy/*.cpp)
list(REMOVE_ITEM STOCKFISH_SOURCES ${STOCKFISH}/main.cpp)
add_library(chess69_stockfish SHARED src/main/cpp/stockfish-jni.cpp ${STOCKFISH_SOURCES})
target_compile_features(chess69_stockfish PRIVATE cxx_std_17)
target_compile_definitions(chess69_stockfish PRIVATE NDEBUG USE_NEON IS_64BIT)
target_compile_options(chess69_stockfish PRIVATE -O3 -fexceptions -frtti -Wa,-I${STOCKFISH})
target_include_directories(chess69_stockfish PRIVATE ${STOCKFISH})
target_link_options(chess69_stockfish PRIVATE -Wl,-z,max-page-size=16384)

file(SHA256 ${STOCKFISH}/nn-1c0000000000.nnue BIG_NET_SHA)
file(SHA256 ${STOCKFISH}/nn-37f18f62d772.nnue SMALL_NET_SHA)
if(NOT BIG_NET_SHA STREQUAL "1c0000000000a67d629999d932d0c373f7450ce43cd12d0562868f4eaf9ae2ad" OR NOT SMALL_NET_SHA STREQUAL "37f18f62d772f3107e1d6aaca3898c130c3c86f2ab63e6555fbbca20635a899d")
  message(FATAL_ERROR "Stockfish NNUE integrity check failed")
endif()
